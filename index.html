<script>

let playerScore = 0;

let computerScore = 0;

let playing = false;

let waiting = false;

const playerFace = document.getElementById("playerFace");

const computerFace = document.getElementById("computerFace");

const playerArrow = document.getElementById("playerArrow");

const computerArrow = document.getElementById("computerArrow");

const statusText = document.getElementById("status");

const message = document.getElementById("message");

const leftBtn = document.getElementById("leftBtn");

const rightBtn = document.getElementById("rightBtn");

const startBtn = document.getElementById("startBtn");

let computerDirection = "";

function startGame() {

    if (playing) return;

    playing = true;

    waiting = false;

    playerScore = 0;

    computerScore = 0;

    updateScore();

    startBtn.disabled = true;

    leftBtn.disabled = false;

    rightBtn.disabled = false;

    playerArrow.textContent = "❔";

    computerArrow.textContent = "❔";

    statusText.textContent = "준비하세요!";

    message.textContent = "공격 방향을 선택하세요!";

}

function attack(playerDirection) {

    if (!playing || waiting) return;

    waiting = true;

    leftBtn.disabled = true;

    rightBtn.disabled = true;

    statusText.textContent = "참! 참! 참!";

    playerFace.className = "face " + playerDirection;

    playerArrow.textContent =

        playerDirection === "left" ? "⬅️" : "➡️";

    // 컴퓨터가 랜덤 방향으로 고개를 돌림

    computerDirection =

        Math.random() < 0.5 ? "left" : "right";

    computerFace.className =

        "face " + computerDirection;

    computerArrow.textContent =

        computerDirection === "left" ? "⬅️" : "➡️";

    setTimeout(() => {

        if (playerDirection !== computerDirection) {

            playerScore++;

            statusText.textContent = "🎉 플레이어 승리!";

            message.textContent =

                "컴퓨터가 공격을 피하지 못했어요!";

        } else {

            computerScore++;

            statusText.textContent = "🤖 컴퓨터 승리!";

            message.textContent =

                "컴퓨터가 공격을 막았어요!";

        }

        updateScore();

        setTimeout(() => {

            if (!playing) return;

            playerFace.className = "face";

            computerFace.className = "face";

            playerArrow.textContent = "❔";

            computerArrow.textContent = "❔";

            statusText.textContent = "다음 공격!";

            message.textContent =

                "왼쪽 또는 오른쪽을 선택하세요!";

            waiting = false;

            leftBtn.disabled = false;

            rightBtn.disabled = false;

        }, 1300);

    }, 500);

}

function updateScore() {

    document.getElementById("playerScore").textContent =

        playerScore;

    document.getElementById("computerScore").textContent =

        computerScore;

}

function resetGame() {

    playing = false;

    waiting = false;

    playerScore = 0;

    computerScore = 0;

    updateScore();

    playerFace.className = "face";

    computerFace.className = "face";

    playerArrow.textContent = "❔";

    computerArrow.textContent = "❔";

    statusText.textContent = "게임을 시작하세요!";

    message.textContent = "플레이어가 공격 방향을 선택하세요!";

    leftBtn.disabled = true;

    rightBtn.disabled = true;

    startBtn.disabled = false;

}
